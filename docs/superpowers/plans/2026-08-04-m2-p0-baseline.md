```python
"""Embedding 模块 — bge-small-zh-v1.5 封装

提供文本向量化能力，用于 ChromaDB 索引和查询。
bge-small-zh-v1.5: 512 维，中文检索效果好，体积约 100MB。
"""
from __future__ import annotations

import numpy as np
from sentence_transformers import SentenceTransformer

# bge 查询指令前缀（官方推荐）
QUERY_INSTRUCTION = "为这个句子生成表示以用于检索相关文章："


class Embedder:
    """bge-small-zh-v1.5 嵌入器

    线程安全：模型加载后只读，可并发调用。
    首次加载会从 HuggingFace 下载模型（~100MB），后续从缓存加载。
    """

    def __init__(self, model_name: str = "BAAI/bge-small-zh-v1.5") -> None:
        self._model = SentenceTransformer(model_name)
        self._dim = self._model.get_sentence_embedding_dimension()

    @property
    def dimension(self) -> int:
        """嵌入维度"""
        return self._dim

    def embed(self, text: str) -> list[float]:
        """嵌入单条文本（用于文档索引）"""
        vec = self._model.encode(text, normalize_embeddings=True)
        return vec.tolist()

    def embed_batch(self, texts: list[str]) -> list[list[float]]:
        """批量嵌入文本（用于文档索引）"""
        vecs = self._model.encode(texts, normalize_embeddings=True)
        return vecs.tolist()

    def embed_query(self, query: str) -> list[float]:
        """嵌入查询文本（添加 bge 查询指令前缀）"""
        # bge 官方建议查询时加前缀指令提升检索效果
        formatted = QUERY_INSTRUCTION + query
        vec = self._model.encode(formatted, normalize_embeddings=True)
        return vec.tolist()
```

```python
"""向量检索模块 — ChromaDB 封装

提供文档入库、向量检索能力。
ChromaDB 嵌入式模式，无需独立进程，2 核 4G 够用。
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any

import chromadb
from chromadb.api.types import EmbeddingFunction

from src.rag.embed import Embedder


@dataclass
class RetrievalResult:
    """单条检索结果"""
    id: str
    text: str
    metadata: dict[str, Any]
    distance: float


class BGEEmbeddingFunction(EmbeddingFunction):
    """ChromaDB 兼容的 bge 嵌入函数

    ChromaDB 要求 EmbeddingFunction 实现 __call__。
    文档嵌入不加指令前缀，查询嵌入加前缀。
    """

    def __init__(self, embedder: Embedder) -> None:
        self._embedder = embedder

    def __call__(self, input: list[str]) -> list[list[float]]:
        return self._embedder.embed_batch(input)

    def embed_query(self, query: str) -> list[float]:
        return self._embedder.embed_query(query)

    # ChromaDB 0.5+ 需要此方法
    def name(self) -> str:
        return "bge-small-zh-v1.5"


class VectorStore:
    """ChromaDB 向量库

    用法：
        store = VectorStore("data/processed/chroma_db")
        store.add_documents([{"id": "c1", "text": "...", "metadata": {...}}])
        results = store.search("查询文本", top_k=5)
    """

    def __init__(
        self,
        persist_path: str,
        collection_name: str = "shanghan_clauses",
        embedder: Embedder | None = None,
    ) -> None:
        self._embedder = embedder or Embedder()
        self._ef = BGEEmbeddingFunction(self._embedder)
        self._client = chromadb.PersistentClient(path=persist_path)
        self._collection = self._client.get_or_create_collection(
            name=collection_name,
            embedding_function=self._ef,
            metadata={"hnsw:space": "cosine"},
        )

    def add_documents(self, documents: list[dict]) -> None:
        """批量添加文档

        Args:
            documents: 每个含 id, text, metadata
        """
        ids = [d["id"] for d in documents]
        texts = [d["text"] for d in documents]
        metadatas = [d.get("metadata", {}) for d in documents]
        self._collection.add(
            ids=ids,
            documents=texts,
            metadatas=metadatas,
        )

    def search(self, query: str, top_k: int = 5) -> list[RetrievalResult]:
        """向量检索

        Args:
            query: 查询文本
            top_k: 返回结果数

        Returns:
            按相似度降序排列的检索结果列表
        """
        if self._collection.count() == 0:
            return []

        query_embedding = self._ef.embed_query(query)
        results = self._collection.query(
            query_embeddings=[query_embedding],
            n_results=min(top_k, self._collection.count()),
        )

        retrieval_results = []
        if results["ids"] and results["ids"][0]:
            for i, doc_id in enumerate(results["ids"][0]):
                retrieval_results.append(RetrievalResult(
                    id=doc_id,
                    text=results["documents"][0][i],
                    metadata=results["metadatas"][0][i],
                    distance=results["distances"][0][i],
                ))
        return retrieval_results

    def count(self) -> int:
        """返回库中文档数"""
        return self._collection.count()

    def clear(self) -> None:
        """清空集合"""
        self._client.delete_collection(self._collection.name)
        self._collection = self._client.get_or_create_collection(
            name=self._collection.name,
            embedding_function=self._ef,
            metadata={"hnsw:space": "cosine"},
        )
```

```python
"""生成模块 — Ollama 推理 + Prompt 构建

通过 Ollama 调用 Qwen2.5-1.5B-Instruct 进行推理。
支持同步生成和流式生成。
"""
from __future__ import annotations

from typing import Iterator

import ollama

from src.rag.retrieve import RetrievalResult

# P0 system prompt（不含风格模仿，遵循 §13.4 设计决策）
SYSTEM_PROMPT = """你是一位中医老师，擅长用通俗易懂的方式讲解中医经典知识。
请根据提供的经典原文回答问题。

注意事项：
- 引用经典原文时标注条文编号
- 解释方剂时列出完整组成
- 不提供具体诊疗建议
- 如果检索结果中没有相关信息，请如实说明"""


def format_retrieved_docs(docs: list[RetrievalResult]) -> str:
    """格式化检索结果为上下文文本"""
    if not docs:
        return "（未检索到相关经典原文）"

    lines = []
    for i, doc in enumerate(docs, 1):
        clause_num = doc.metadata.get("clause_number", "?")
        chapter = doc.metadata.get("chapter", "")
        header = f"{i}. （第{clause_num}条"
        if chapter:
            header += f"·{chapter}"
        header += f"）{doc.text}"
        lines.append(header)
    return "\n".join(lines)


def build_prompt(question: str, docs: list[RetrievalResult]) -> str:
    """构建用户消息（system + context + question）"""
    context = format_retrieved_docs(docs)
    return f"""{SYSTEM_PROMPT}

[检索到的经典原文]
{context}

[用户问题]
{question}"""


class Generator:
    """Ollama 推理封装

    用法：
        gen = Generator(model="qwen2.5:1.5b")
        answer = gen.generate("桂枝汤的组成", retrieved_docs)
    """

    def __init__(self, model: str = "qwen2.5:1.5b") -> None:
        self._model = model

    def generate(
        self,
        question: str,
        docs: list[RetrievalResult],
        temperature: float = 0.7,
        max_tokens: int = 512,
    ) -> str:
        """同步生成回答

        Args:
            question: 用户问题
            docs: 检索到的文档
            temperature: 生成温度
            max_tokens: 最大生成 token 数

        Returns:
            模型生成的回答文本
        """
        prompt = build_prompt(question, docs)
        try:
            response = ollama.chat(
                model=self._model,
                messages=[
                    {"role": "system", "content": SYSTEM_PROMPT},
                    {"role": "user", "content": f"[检索到的经典原文]\n{format_retrieved_docs(docs)}\n\n[用户问题]\n{question}"},
                ],
                options={
                    "temperature": temperature,
                    "num_predict": max_tokens,
                },
            )
            return response["message"]["content"]
        except Exception as e:
            return f"抱歉，生成回答时出现错误：{e}"

    def stream_generate(
        self,
        question: str,
        docs: list[RetrievalResult],
        temperature: float = 0.7,
        max_tokens: int = 512,
    ) -> Iterator[str]:
        """流式生成回答

        Args:
            question: 用户问题
            docs: 检索到的文档

        Yields:
            生成文本的每个 chunk
        """
        try:
            for chunk in ollama.chat(
                model=self._model,
                messages=[
                    {"role": "system", "content": SYSTEM_PROMPT},
                    {"role": "user", "content": f"[检索到的经典原文]\n{format_retrieved_docs(docs)}\n\n[用户问题]\n{question}"},
                ],
                options={
                    "temperature": temperature,
                    "num_predict": max_tokens,
                },
                stream=True,
            ):
                content = chunk.get("message", {}).get("content", "")
                if content:
                    yield content
        except Exception as e:
            yield f"生成回答时出现错误：{e}"
```

```python
"""评测指标模块

计算 P0/P1/P2 阶段通用的评测指标。
所有指标可自动计算，不依赖人工判断。
"""
from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Any


@dataclass
class EvalResult:
    """单题评测结果"""
    question_id: int
    category: str
    question: str
    answer: str
    expected_answer: str
    reference_clauses: list[int]
    retrieved_clauses: list[int]
    latency: float


def compute_recall_at_k(results: list[EvalResult], k: int = 5) -> float:
    """计算 recall@k

    只统计有 reference_clauses 的题目。
    正确文档（reference_clauses 中任一）出现在 retrieved_clauses 前 k 个中即算命中。

    Returns:
        0.0-1.0 的命中率
    """
    valid = [r for r in results if r.reference_clauses]
    if not valid:
        return 0.0

    hits = 0
    for r in valid:
        top_k = r.retrieved_clauses[:k]
        if any(ref in top_k for ref in r.reference_clauses):
            hits += 1
    return hits / len(valid)


def compute_keyword_accuracy(results: list[EvalResult]) -> float:
    """计算关键词命中率

    将 expected_answer 按标点分词，检查每个关键词是否出现在 answer 中。
    忽略单字词和通用标点。

    Returns:
        0.0-1.0 的平均命中率
    """
    scores = []
    for r in results:
        # 提取 expected_answer 中的关键词（2 字以上）
        keywords = re.findall(r"[\u4e00-\u9fff]{2,}", r.expected_answer)
        # 去重
        keywords = list(dict.fromkeys(keywords))
        if not keywords:
            scores.append(1.0)
            continue

        hits = sum(1 for kw in keywords if kw in r.answer)
        scores.append(hits / len(keywords))

    return sum(scores) / len(scores) if scores else 0.0


def compute_latency_stats(results: list[EvalResult]) -> dict[str, float]:
    """计算延迟统计

    Returns:
        dict 含 min, max, mean, p50, p95
    """
    if not results:
        return {"min": 0, "max": 0, "mean": 0, "p50": 0, "p95": 0}

    latencies = sorted(r.latency for r in results)
    n = len(latencies)

    def percentile(p: float) -> float:
        idx = max(0, min(n - 1, int(n * p) - 1))
        return latencies[idx]

    return {
        "min": latencies[0],
        "max": latencies[-1],
        "mean": round(sum(latencies) / n, 2),
        "p50": round(percentile(0.50), 2),
        "p95": round(percentile(0.95), 2),
    }


def compute_category_accuracy(results: list[EvalResult]) -> dict[str, float]:
    """按类别计算关键词命中率

    Returns:
        {category: accuracy} 字典
    """
    from collections import defaultdict

    by_category: dict[str, list[EvalResult]] = defaultdict(list)
    for r in results:
        by_category[r.category].append(r)

    return {
        cat: round(compute_keyword_accuracy(items), 4)
        for cat, items in by_category.items()
    }


def generate_report(
    results: list[EvalResult],
    stage: str = "P0",
) -> dict[str, Any]:
    """生成完整评测报告

    Args:
        results: 所有题目的评测结果
        stage: 阶段标识（P0/P1/P2）

    Returns:
        完整报告字典
    """
    return {
        "stage": stage,
        "total_questions": len(results),
        "recall_at_5": round(compute_recall_at_k(results, k=5), 4),
        "keyword_accuracy": round(compute_keyword_accuracy(results), 4),
        "category_accuracy": compute_category_accuracy(results),
        "latency": compute_latency_stats(results),
    }
```

```python
"""RAG 管线编排 — 检索 → 生成

将向量检索和生成模块组合为完整的 RAG 管线。
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Iterator

from src.rag.embed import Embedder
from src.rag.generate import Generator
from src.rag.retrieve import VectorStore, RetrievalResult


@dataclass
class RAGResponse:
    """RAG 完整响应"""
    answer: str
    retrieved_docs: list[RetrievalResult]
    latency: float


class RAGPipeline:
    """RAG 管线

    用法：
        pipeline = RAGPipeline(chroma_path="data/processed/chroma_db")
        response = pipeline.query("桂枝汤的组成")
        print(response.answer)
    """

    def __init__(
        self,
        chroma_path: str = "data/processed/chroma_db",
        model: str = "qwen2.5:1.5b",
        top_k: int = 5,
    ) -> None:
        self._embedder = Embedder()
        self._store = VectorStore(
            persist_path=chroma_path,
            embedder=self._embedder,
        )
        self._generator = Generator(model=model)
        self._top_k = top_k

    def retrieve(self, question: str) -> list[RetrievalResult]:
        """仅检索，不生成"""
        return self._store.search(question, top_k=self._top_k)

    def query(
        self,
        question: str,
        temperature: float = 0.7,
        max_tokens: int = 512,
    ) -> RAGResponse:
        """完整 RAG 查询：检索 → 生成"""
        import time
        start = time.time()

        docs = self.retrieve(question)
        answer = self._generator.generate(
            question, docs,
            temperature=temperature,
            max_tokens=max_tokens,
        )

        return RAGResponse(
            answer=answer,
            retrieved_docs=docs,
            latency=round(time.time() - start, 2),
        )

    def stream_query(
        self,
        question: str,
        temperature: float = 0.7,
        max_tokens: int = 512,
    ) -> tuple[list[RetrievalResult], Iterator[str]]:
        """流式 RAG 查询：先检索，再流式生成"""
        docs = self.retrieve(question)
        stream = self._generator.stream_generate(
            question, docs,
            temperature=temperature,
            max_tokens=max_tokens,
        )
        return docs, stream

    @property
    def store(self) -> VectorStore:
        """暴露向量库（用于统计/管理）"""
        return self._store
```

```python
"""评测运行器

编排 RAG 管线跑 50 题评测集，生成评测报告。
"""
from __future__ import annotations

import json
import time
from typing import Any

from src.eval.dataset import load_eval_set
from src.eval.metrics import (
    EvalResult,
    generate_report,
)


class EvalRunner:
    """评测运行器

    用法：
        runner = EvalRunner(pipeline, "data/eval/eval_50.jsonl")
        results = runner.run()
        report = runner.generate_report(results, stage="P0")
        runner.save_report(results, "data/processed/p0_report.json", stage="P0")
    """

    def __init__(self, pipeline: Any, eval_path: str) -> None:
        self._pipeline = pipeline
        self._questions = load_eval_set(eval_path)

    def run(self) -> list[EvalResult]:
        """运行全部评测题目"""
        results: list[EvalResult] = []
        for q in self._questions:
            print(f"  [{q.id}/{len(self._questions)}] {q.category}: {q.question[:30]}...")

            response = self._pipeline.query(q.question)
            retrieved_clauses = [
                doc.metadata.get("clause_number", 0)
                for doc in response.retrieved_docs
            ]

            results.append(EvalResult(
                question_id=q.id,
                category=q.category,
                question=q.question,
                answer=response.answer,
                expected_answer=q.expected_answer,
                reference_clauses=q.reference_clauses,
                retrieved_clauses=retrieved_clauses,
                latency=response.latency,
            ))

            print(f"    → 延迟 {response.latency}s, 检索到 {len(retrieved_clauses)} 条")

        return results

    def generate_report(self, results: list[EvalResult], stage: str = "P0") -> dict[str, Any]:
        """生成评测报告"""
        return generate_report(results, stage=stage)

    def save_report(
        self,
        results: list[EvalResult],
        output_path: str,
        stage: str = "P0",
    ) -> dict[str, Any]:
        """保存评测报告到文件"""
        report = self.generate_report(results, stage=stage)

        # 附带详细结果（供人工审查）
        report["details"] = [
            {
                "question_id": r.question_id,
                "category": r.category,
                "question": r.question,
                "answer": r.answer,
                "expected_answer": r.expected_answer,
                "reference_clauses": r.reference_clauses,
                "retrieved_clauses": r.retrieved_clauses,
                "latency": r.latency,
            }
            for r in results
        ]

        with open(output_path, "w", encoding="utf-8") as f:
            json.dump(report, f, ensure_ascii=False, indent=2)

        print(f"\n评测报告已保存: {output_path}")
        print(f"  阶段: {stage}")
        print(f"  题数: {report['total_questions']}")
        print(f"  recall@5: {report['recall_at_5']}")
        print(f"  关键词命中率: {report['keyword_accuracy']}")
        print(f"  延迟 p50: {report['latency']['p50']}s, p95: {report['latency']['p95']}s")

        return report
```

```python
"""FastAPI Web 服务

提供 SSE 流式对话、健康检查、图谱统计接口。
并发限制为 1（2 核 4G 服务器约束）。
"""
from __future__ import annotations

import asyncio
import json
from pathlib import Path

from fastapi import FastAPI, HTTPException
from fastapi.responses import HTMLResponse, StreamingResponse
from pydantic import BaseModel

from src.data.kg_build import build_graph, graph_stats


class ChatRequest(BaseModel):
    question: str


def create_app(pipeline) -> FastAPI:
    """创建 FastAPI 应用

    Args:
        pipeline: RAGPipeline 实例
    """
    app = FastAPI(title="中医经典知识检索与理解助手", version="0.2.0")

    # 全局并发锁（2 核 4G 约束：同时只处理 1 个请求）
    semaphore = asyncio.Semaphore(1)

    # 预加载知识图谱
    kg_graph = build_graph()
    kg_stats = graph_stats(kg_graph)

    @app.get("/", response_class=HTMLResponse)
    async def index():
        """返回前端页面"""
        html_path = Path(__file__).parent / "frontend.html"
        return HTMLResponse(html_path.read_text(encoding="utf-8"))

    @app.get("/api/health")
    async def health():
        """健康检查"""
        return {
            "status": "ok",
            "vector_store_count": pipeline.store.count(),
            "model": "qwen2.5:1.5b",
        }

    @app.get("/api/graph/stats")
    async def graph_stats_endpoint():
        """图谱统计"""
        return kg_stats

    @app.post("/api/chat")
    async def chat(req: ChatRequest):
        """流式对话（SSE）"""
        if not req.question.strip():
            raise HTTPException(status_code=400, detail="问题不能为空")

        async def generate():
            # 获取并发锁
            async with semaphore:
                # 检索
                docs, stream = pipeline.stream_query(req.question)

                # 先发送检索结果
                retrieved = [
                    {
                        "clause_number": d.metadata.get("clause_number"),
                        "chapter": d.metadata.get("chapter", ""),
                        "text": d.text,
                        "distance": d.distance,
                    }
                    for d in docs
                ]
                yield f"data: {json.dumps({'type': 'retrieved', 'docs': retrieved}, ensure_ascii=False)}\n\n"

                # 流式发送生成内容
                for chunk in stream:
                    yield f"data: {json.dumps({'type': 'chunk', 'content': chunk}, ensure_ascii=False)}\n\n"

                # 结束标记
                yield f"data: {json.dumps({'type': 'done'})}\n\n"

        return StreamingResponse(
            generate(),
            media_type="text/event-stream",
            headers={
                "Cache-Control": "no-cache",
                "Connection": "keep-alive",
            },
        )

    return app
```
